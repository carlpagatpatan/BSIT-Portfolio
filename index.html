Below is the complete source code for my console-based game developed as part of my CC102 requirement:


import java.util.Scanner;
import java.util.Random;

public class ConsoleGamePeTa {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        Random random = new Random();
        
        System.out.println("========================================");
        System.out.println("   WELCOME TO THE CC102 CONSOLE GAME    ");
        System.out.println("========================================");
        
        int targetNumber = random.nextInt(10) + 1;
        int guess = 0;
        int attempts = 0;
        
        System.out.print("Guess a number between 1 and 10: ");
        
        while (guess != targetNumber) {
            if (scanner.hasNextInt()) {
                guess = scanner.nextInt();
                attempts++;
                
                if (guess < targetNumber) {
                    System.out.print("Too low! Try again: ");
                } else if (guess > targetNumber) {
                    System.out.print("Too high! Try again: ");
                } else {
                    System.out.println("Congratulations! You guessed the number in " + attempts + " attempts.");
                }
            } else {
                System.out.print("Invalid input. Please enter a valid number: ");
                scanner.next();
            }
        }
        
        scanner.close();
    }
}
